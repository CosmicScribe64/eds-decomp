# Proposed names from the program-flow / sound / RAM analysis (2026-09-29).
# Format: 0xADDRESS name  # reason.  Style: pret-style CamelCase functions, gCamelCase globals.
# All behaviour was verified by disassembly unless the reason says "hyp".
# See wiki/game/program-flow.md, wiki/rom/ram-map.md, wiki/game/sound-engine.md and wiki/functions/*.md.

## --- boot / main loop / system ---
0x08075DF4 GameInit  # AgbMain's first call: clears EWRAM/IWRAM, ReadSram, fills IntrTable, copies IntrMain to IWRAM, IE/WAITCNT, SoundInit
0x08075D70 MainLoop  # AgbMain's second call: waits for the VBlank flag, FrameSyncUpdate, calls gMain.callback, falls back to the main menu
0x08075CB4 FrameSyncUpdate  # after VBlank: BG scroll from shadows, BG map buffer -> VRAM, FlushOamBuffer, ReadKeys, SoundMain, Random
0x08075C44 FlushOamBuffer  # if vblankFlags&1: copy gMain.oamBuffer -> OAM, reset the count and entries
0x08075228 ReadKeys  # only KEYINPUT reader; held/new/prev keys + D-pad auto-repeat after 20 frames
0x08076F9C Random  # LCG x*0x343FD+0x269EC3, rotate 16, returns 15 bits; called once per frame + 75 other sites
0x080754F8 SetMainCallback  # SaveGame, clears the VBlank cbs/HBlank, zeroes the sequencer bytes, sets gMain.callback
0x080754BC SaveGame  # UpdateSaveChecksum + WriteSram/VerifySram up to 32 tries
0x080755A0 SaveAndResetSceneState  # SaveGame + reset scene state, returns 1; no callers (dead code)
0x08075D6C DebugHook_Nop  # empty (bx lr), called every frame by MainLoop; hyp: stripped debug hook

## --- interrupt handlers ---
0x0807569C VBlankIntr  # IntrTable[2]: sets intrCheck bit0, counters, early cb, SoundVBlank, vblankCallback
0x0807570C Timer2Intr  # IntrTable[6]: reload 0xF400, ++u16 at 0x030051FC
0x08075740 GamepakIntr  # IntrTable[13]: while(1); unreachable (IntrMain loops on GAMEPAK itself)
0x0807E324 SoundDma1Intr  # IntrTable[9]: FIFO ring position +16, rewinds DMA1/DMA2 at 0x2C0
0x08075F74 LinkSerialIntr  # serial+timer3 IRQ (installed by sub_080735D4); reads SIOMULTI, 0xFEFE sync; hyp Nintendo MultiSio-derived
0x080735D4 LinkSioInit  # sets SIO multi-player 115200, installs LinkSerialIntr into 2 IntrTable slots, IE |= SERIAL|TIMER3

## --- fades / video / memory helpers ---
0x080757F4 ClearBlend  # brightness level 0, BLDCNT=0, BLDY=0
0x080759F4 SetBrightnessBlack  # BLDCNT=0x3FFF, BLDY=0x1F
0x08075A30 SetBrightnessWhite  # BLDCNT=0x3FBF, BLDY=0x1F
0x08075A6C FadeToBlack  # level += step, returns 1 at full
0x08075AE4 FadeFromBlack  # level -= step, ClearBlend + return 1 at 0
0x08075B58 FadeToWhite  # like FadeToBlack with BLDCNT 0x3FBF
0x08075BD0 FadeFromWhite  # like FadeFromBlack with BLDCNT 0x3FBF
0x08075278 MemClear16  # zero (size+1)/2 halfwords
0x08075294 MemCopy16  # copy (size+1)/2 halfwords
0x080752B0 CopyDoubleWords  # ldm/stm 8-byte block copy, (size+7)/8 blocks
0x080753CC StrLen  # byte strlen
0x08075630 LoadSystemFontGfx  # 16-color palette 0x0822C300 + 16 tiles 0x0822C320 to BG and OBJ
0x08075744 ResetBgHofs  # zero the shadow + BGxHOFS
0x08075778 ResetBgVofs  # zero the shadow + BGxVOFS
0x080757AC ResetBgScroll  # ResetBgHofs + ResetBgVofs
0x08073498 ClearBgMapBuffers  # clears gMain.bgMapBuffer (16 KiB) + 0x02010014 (0x1C00)
0x08073574 ResetVideo  # ClearBgMapBuffers, LoadSystemFontGfx, identity BG2/3 affine, BG0CNT=4
0x08072FAC LoadBgImage  # (mapBase, palIdx, tileBase, img): palette + 8bpp tiles to 0x06004000 + map into gMain.bgMapBuffer
0x080761F0 AddSprite  # append an OAM entry to gMain.oamBuffer[gMain.oamCount++]

## --- game-side sound API ---
0x08077A44 IsSeEnabled  # gSaveData+0x2152 bit0
0x08077A5C IsBgmEnabled  # gSaveData+0x2152 bit1
0x08077A74 SetSeEnabled  # set/clear bit0
0x08077AB0 SetBgmEnabled  # set/clear bit1
0x08077AEC PlaySE  # gated on SE flag + one SE per frame (gMain.lastSeFrame); 288 call sites

## --- Konami sound driver ---
0x0807D578 SoundInit  # SOUNDCNT/BIAS/PSG init, driver struct init, wave RAM, SoundDmaInit
0x0807D3D0 SoundDmaInit  # Timer0 0xFCB9 (~20 kHz), DMA1/DMA2 -> FIFO A/B, copies mix loop to IWRAM 0x03005A54
0x0807D518 SoundLoadWaveRam  # 16-byte wave pattern from 0x08139550 -> WAVE_RAM, bank toggle
0x0807DB58 SoundSequencerTick  # per-VBlank sequencer for 10 BGM + 6 SE tracks, writes PSG regs
0x0807E3B0 SoundVBlank  # unless paused: SoundSequencerTick + ARM mixer (veneer 0x08080A18)
0x0807E3D8 SoundStartPendingSE  # starts the SE requested in drv+0x18C (SE table 0x08087FD0)
0x0807E554 SoundMain  # main-thread: pending SE + BGM start from the song table 0x080E09D0
0x0807E674 SoundRequestBGM  # drv+0x18A = id
0x0807E6B0 SoundIsBGMPlaying  # (id)
0x0807E764 SoundFadeOutBGM  # (speed)
0x0807E814 SoundRequestSE  # drv+0x18C = id (bit15: don't restart if playing)
0x0807E918 SoundPcmStart  # set up a PCM voice from the sample tables 0x0811B420/0x08088A20 + pitch table 0x081A960C
0x0807E984 SoundCountActivePcm  # counts active PCM voices
0x0807EAA0 SoundStopBGM  # drv flags |= 1
0x0807EAD0 SoundMixAll  # ARM: mixes voices 0-2 -> FIFO A buffer, 3-5 -> FIFO B buffer
0x0807EAF0 SoundMixFifo  # ARM: clear + mix one FIFO ring span
0x0807EC1C SoundMixChannel  # ARM inner loop; copied to and executed at IWRAM 0x03005A54
